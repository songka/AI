33class BoundingBox(BaseModel): 

34 """3D bounding box of the part in mm.""" 

35 

36 min_x: float 

37 min_y: float 

38 min_z: float = 0.0 

39 max_x: float 

40 max_y: float 

41 max_z: float = 0.0 

42 

43 @property 

44 def length(self) -> float: 

45 """X-axis span.""" 

46 return self.max_x - self.min_x 

47 

48 @property 

49 def width(self) -> float: 

50 """Y-axis span.""" 

51 return self.max_y - self.min_y 

52 

53 @property 

54 def height(self) -> float: 

55 """Z-axis span."""

182 # -- Computation helpers -- 

183 def calculate_weight(self, density_g_cm3: float) -> float | None: 

184 """Calculate weight from volume and material density. 

185 

186 Args: 

187 density_g_cm3: Material density in g/cm³. 

188 

189 Returns: 

190 Weight in kg, or None if volume is not available. 

191 """ 

192 if self.volume_mm3 is None: 

193 return None 

194 # volume_mm3 / 1000 → cm³; cm³ × density → g; g / 1000 → kg 

195 return round((self.volume_mm3 / 1000.0) * density_g_cm3 / 1000.0, 3)
