81 """Check API connectivity. Returns health status without exposing key.""" 

82 if not self.is_configured: 

83 return {"configured": False, "reachable": False, "error": "API key not configured"} 

84 

85 t0 = time.monotonic() 

86 last_error = "未知连接错误" 

87 for attempt in range(2): 

88 try: 

89 with httpx.Client(timeout=10.0) as client: 

90 resp = client.get( 

91 f"{self._base_url}/models", 

92 headers=self._headers(), 

93 ) 

94 latency_ms = (time.monotonic() - t0) * 1000 

95 if resp.status_code == 200: 

96 data = resp.json() 

97 models = [m.get("id", "") for m in data.get("data", [])] 

98 model_found = any(self._model in m for m in models) 

99 return { 

100 "configured": True, 

101 "reachable": True, 

102 "model": self._model, 

103 "model_found": model_found, 

104 "latency_ms": round(latency_ms, 1), 

105 "error": None, 

106 } 

107 last_error = f"HTTP {resp.status_code}" 

108 if resp.status_code >= 500 and attempt == 0: 

109 time.sleep(0.5) 

110 continue 

111 break 

112 except httpx.TimeoutException: 

113 self.last_error = f"AI服务响应超时（{self._timeout:g}秒）" 

114 self._circuit_open_until = time.monotonic() + 30.0 

115 return None 

116 except httpx.RequestError as exc: 

117 last_error = str(exc) 

118 if attempt == 0: 

119 time.sleep(0.5) 

120 continue 

121 break 

122 except Exception as exc: 

123 last_error = str(exc) 

124 break 

125 latency_ms = (time.monotonic() - t0) * 1000 

126 return { 

127 "configured": True, 

128 "reachable": False, 

129 "model": self._model, 

130 "latency_ms": round(latency_ms, 1),

228 "LATHE": "車床", 

229 "MILL": "銑床", 

230 "GRIND": "磨床", 

231 "FITTER": "鉗工",

238只能使用以下代码：{', '.join(allowed)}。不要输出材料、表面处理、焊接或装配；没有充分依据就不要选择。

293 def _simple_part_hour_limit( 

294 code: str, 

295 texts: list[str], 

296 geometry_summary: dict[str, Any], 

297 ) -> float | None: 

298 """Bound unsupported AI hours for a geometrically simple single part.""" 

299 context = " ".join(str(text) for text in texts).lower() 

300 explicit_patterns = { 

301 "CNC": r"cnc|数控|數控|加工中心|多轴|多軸|曲面", 

302 "LATHE": r"车削|車削|车床|車床|回转体|回轉體|轴类|軸類|套筒", 

303 "MILL": r"铣削|銑削|铣床|銑床|深腔|复杂轮廓|複雜輪廓", 

304 "GRIND": r"磨削|磨床|平面度|圆度|圓度|ra\s*[0-1](?:\.|。)", 

305 "FITTER": r"钳工|鉗工|去毛刺|修配|铰孔|鉸孔", 

306 "EDM": r"edm|放电|放電|电火花|電火花|内尖角|內尖角", 

307 "WIRE_CUT": r"线切割|線切割|快丝|快絲|wire\s*cut", 

308 "SLOW_WIRE": r"慢丝|慢絲|精密线切割|精密線切割", 

309 } 

310 if re.search(explicit_patterns.get(code, r"$^"), context, re.IGNORECASE): 

311 return None 

312 try: 

313 holes = max(float(geometry_summary.get("孔数量", 0) or 0), 0) 

314 threads = max(float(geometry_summary.get("螺纹数量", 0) or 0), 0) 

315 entities = max(float(geometry_summary.get("几何实体数量", 0) or 0), 0) 

316 dimensions = geometry_summary.get("外形尺寸") or [] 

317 span = max((float(value) for value in dimensions), default=0) 

318 except (TypeError, ValueError): 

319 return 1.5 

320 baseline = ( 

321 0.5 

322 + holes * 0.08 

323 + threads * 0.05 

324 + min(span / 1000.0, 0.75) 

325 + min(entities / 200.0, 0.75) 

326 ) 

327 return round(min(max(baseline * 1.75, 1.0), 3.0), 2)

532 def _chat( 

533 self, 

534 messages: list[dict[str, str]], 

535 temperature: float = 0.1, 

536 max_tokens: int | None = None, 

537 ) -> str | None: 

538 """Send a chat request. Returns content string or None.""" 

539 payload: dict[str, Any] = { 

540 "model": self._model, 

541 "messages": messages, 

542 "temperature": temperature, 

543 "max_tokens": max_tokens or self._max_tokens, 

544 } 

545 cacheable = temperature <= 0.1 

546 cache_key = hashlib.sha256( 

547 json.dumps( 

548 { 

549 "schema": 1, 

550 "base_url": self._base_url, 

551 **payload, 

552 }, 

553 ensure_ascii=False, 

554 sort_keys=True, 

555 separators=(",", ":"), 

556 ).encode("utf-8") 

557 ).hexdigest() 

558 if cacheable: 

559 with self._cache_lock: 

560 cached = self._response_cache.get(cache_key) 

561 if cached is not None: 

562 self._response_cache.move_to_end(cache_key) 

563 self.cache_hits += 1 

564 self.last_error = None 

565 return cached 

566 self.cache_misses += 1 

567 if time.monotonic() < self._circuit_open_until: 

568 return None 

569 

570 request_payload = {**payload, "thinking": {"type": "disabled"}} 

571 for attempt in range(2): 

572 try: 

573 content = self._send_request(request_payload) 

574 self._circuit_open_until = 0.0 

575 self.last_error = None 

576 if cacheable and content and self._parse_json_object(content) is not None: 

577 with self._cache_lock: 

578 self._response_cache[cache_key] = content 

579 self._response_cache.move_to_end(cache_key) 

580 while len(self._response_cache) > self._cache_limit: 

581 self._response_cache.popitem(last=False) 

582 return content 

583 except httpx.HTTPStatusError as exc: 

584 if exc.response.status_code == 400 and "thinking" in request_payload: 

585 # Compatibility fallback for gateways without the thinking field. 

586 request_payload = payload 

587 continue 

588 if exc.response.status_code >= 500 and attempt == 0: 

589 time.sleep(0.5) 

590 continue 

591 self.last_error = f"AI服务返回 HTTP {exc.response.status_code}" 

592 self._circuit_open_until = time.monotonic() + 30.0 

593 return None 

594 except httpx.RequestError as exc: 

595 if attempt == 0: 

596 time.sleep(0.5) 

597 continue 

598 self.last_error = f"AI服务连接失败：{exc}" 

599 self._circuit_open_until = time.monotonic() + 30.0 

600 return None 

601 except Exception as exc: 

602 self.last_error = f"AI调用失败：{exc}" 

603 self._circuit_open_until = time.monotonic() + 30.0 

604 return None 

605 return None

640 def _parse_extraction_response(self, content: str) -> dict[str, Any] | None: 

641 """Parse AI JSON response into structured dict.""" 

642 try: 

643 # Try direct JSON parse 

644 result = json.loads(content) 

645 except json.JSONDecodeError: 

646 # Try to extract JSON from markdown code blocks 

647 import re 

648 m = re.search(r'```(?:json)?\s*(\{.*?\})\s*```', content, re.DOTALL) 

649 if m: 

650 try: 

651 result = json.loads(m.group(1)) 

652 except json.JSONDecodeError: 

653 return None 

654 else: 

655 # Try to find any JSON object 

656 m = re.search(r'\{.*\}', content, re.DOTALL) 

657 if m: 

658 try: 

659 result = json.loads(m.group(0)) 

660 except json.JSONDecodeError: 

661 return None 

662 else: 

663 return None 

664 

665 return { 

666 "drawing_number": result.get("drawing_number"), 

667 "material_candidate": result.get("material_candidate"), 

668 "surface_treatment_candidate": result.get("surface_treatment_candidate"), 

669 "heat_treatment_candidate": result.get("heat_treatment_candidate"), 

670 "thickness_candidate": result.get("thickness_candidate"), 

671 "missing_fields": result.get("missing_fields", []), 

672 "warnings": result.get("warnings", []), 

673 "confidence": float(result.get("confidence", 0.0)), 

674 }

676 @staticmethod 

677 def _parse_json_object(content: str) -> dict[str, Any] | None: 

678 """Parse a JSON object, tolerating a Markdown code fence.""" 

679 try: 

680 value = json.loads(content) 

681 except json.JSONDecodeError: 

682 import re 

683 match = re.search(r"```(?:json)?\s*(\{.*?\})\s*```", content, re.DOTALL) 

684 if not match: 

685 match = re.search(r"\{.*\}", content, re.DOTALL) 

686 if not match: 

687 return None 

688 try: 

689 value = json.loads(match.group(1) if match.lastindex else match.group(0)) 

690 except json.JSONDecodeError: 

691 return None 

692 return value if isinstance(value, dict) else None
