219def display_price_code(value: Any) -> str: 

220 """Show a user-facing Chinese material/process name while retaining codes internally.""" 

221 if value is None or value == "": 

222 return "—" 

223 text = str(value).strip() 

224 return PRICE_CODE_LABELS.get(text.upper(), PRICE_CODE_LABELS.get(text, text))

227def display_origin_supplier(value: Any) -> str: 

228 """Never expose supplier IDs as the user-facing source name.""" 

229 if value is None or value == "": 

230 return "公司内部核准价" 

231 return ORIGIN_SUPPLIER_LABELS.get(str(value), "供应商名称未维护")

234def display_value(key: str, value: Any) -> str: 

235 if value is None or value == "": 

236 return "—" 

237 text = str(value) 

238 if key in {"canonical_code", "canonical_code_display", "material_code", "target_name"}: 

239 return display_price_code(value) 

240 if key in {"origin_supplier_id", "origin_supplier_name"}: 

241 return display_origin_supplier(value) 

242 if key in {"quotation_status", "status"}: 

243 return STATUS_LABELS.get(text, text) 

244 if key in {"target_type", "category"}: 

245 return TYPE_LABELS.get(text, text) 

246 if key == "field_name": 

247 return FIELD_VALUE_LABELS.get(text, text) 

248 if key == "unit": 

249 return UNIT_LABELS.get(text.casefold(), text) 

250 if key in {"confidence", "ai_estimate_confidence"}: 

251 if key == "ai_estimate_confidence": 

252 try: 

253 return f"{float(value) * 100:.0f}%" 

254 except (TypeError, ValueError): 

255 return text 

256 return CONFIDENCE_LABELS.get(text.casefold(), text) 

257 if key in {"source", "quote_price_source", "origin_price_source"}: 

258 return SOURCE_LABELS.get(text, text) 

259 if key in {"resolution_source", "resolution_display"}: 

260 return RESOLUTION_LABELS.get(text, text) 

261 if key in {"read_only", "fallback_warning", "ai_accepted", "recalculated", "ai_used"}: 

262 return "是" if bool(value) else "否" 

263 if key == "tax_rate": 

264 try: 

265 numeric = float(value) 

266 return f"{numeric * 100:.0f}%" if numeric <= 1 else f"{numeric:.0f}%" 

267 except (TypeError, ValueError): 

268 return text 

269 if key == "cost_completion": 

270 return f"{float(value):.1f}%" 

271 if key in { 

272 "unit_price", 

273 "amount", 

274 "subtotal_excl_tax", 

275 "tax_amount", 

276 "total_incl_tax", 

277 "parsed_value", 

278 "ai_estimated_unit_price", 

279 "ai_estimated_amount", 

280 }: 

281 try: 

282 return f"{float(value):,.2f}" 

283 except (TypeError, ValueError): 

284 return text 

285 if key == "quantity": 

286 try: 

287 return f"{float(value):,.4f}".rstrip("0").rstrip(".") 

288 except (TypeError, ValueError): 

289 return text 

290 if key in {"created_at", "updated_at"}: 

291 return text.replace("T", " ").split("+")[0] 

292 return text
