1"""BOM Description Parser. 

2 

3Parses semicolon-delimited BOM description strings into structured ParsedPart objects. 

4 

5Format: 

6 segment0;segment1;segment2;segment3;segment4;segment5;... 

7 

8Typical patterns: 

9 加工件: 原材料;加工件;S50C;J003;928*796*15;表面鍍鉻 

10 電控件: 原材料;電控外購件;控制類;PLC擴展;擴展IO模塊;型號:AS16AP11T-A;品牌:台達 

11 機構件: 原材料;機構外購件;鋁型材;40*40;圖號:W001;1300*1300*995

31class ParseStatus(str, Enum): 

32 SUCCESS = "success" # All expected fields present 

33 PARTIAL = "partial" # Some fields missing 

34 FAILED = "failed" # Cannot parse at all 

35 

36 

37@dataclass 

38class ParseIssue: 

39 """A single issue found during parsing.""" 

40 severity: str # "warning" | "error" 

41 field: str # Which field is affected 

42 message: str

45@dataclass 

46class DescriptionParseResult: 

47 """Complete result of parsing a BOM description.""" 

48 

49 # The populated ParsedPart (domain model) 

50 parsed_part: ParsedPart 

51 

52 # Parse status 

53 status: ParseStatus = ParseStatus.SUCCESS 

54 

55 # Issues found 

56 issues: list[ParseIssue] = field(default_factory=list)

59 raw_segments: list[str] = field(default_factory=list)

66# Known sub-types and which kind they map to 

67_MACHINED_SUBTYPES = {"加工件"} 

68_ELECTRICAL_SUBTYPES = {"電控外購件", "电控外购件"} 

69_MECHANICAL_SUBTYPES = {"機構外購件", "机构外购件", "機加件"}

72class DescriptionParser: 

73 """Parses BOM description text into structured ParsedPart objects. 

74 

75 Usage: 

76 parser = DescriptionParser() 

77 result = parser.parse(bom_entry) 

78 # result.parsed_part → ParsedPart domain model 

79 # result.status → ParseStatus 

80 # result.issues → list of ParseIssue 

81 """ 

82 

83 def __init__(self, separator: str = ";"): 

84 self._separator = separator 

85 

86 # -- Public API -- 

87 

88 def parse( 

89 self, 

90 bom_item: str, 

91 description: str, 

92 source_row: int = 0, 

93 unit_cost: float = 0.0, 

94 ) -> DescriptionParseResult: 

95 """Parse a single BOM description string. 

96 

97 Args: 

98 bom_item: The BomEntry.item identifier. 

99 description: The full semicolon-delimited description. 

100 source_row: Source Excel row for tracing. 

101 unit_cost: Unit cost from BOM. 

102 

103 Returns: 

104 DescriptionParseResult with populated ParsedPart and status. 

105 """ 

106 issues: list[ParseIssue] = [] 

107 

108 # Split into segments 

109 segments = [s.strip() for s in description.split(self._separator)] 

110 

111 # Build ParsedPart with defaults 

112 pp = ParsedPart( 

113 bom_item=bom_item, 

114 source_row=source_row, 

115 unit_cost=unit_cost, 

116 ) 

117 

118 # Parse category (segment 0) 

119 if len(segments) > 0 and segments[0]: 

120 pp.category = segments[0] 

121 else: 

122 issues.append(ParseIssue("warning", "category", "Missing category (segment 0)")) 

123 

124 # Parse sub_type (segment 1) — determines is_quotable 

125 if len(segments) > 1 and segments[1]: 

126 pp.sub_type = segments[1] 

127 pp = self._classify_part(pp, segments[1]) 

128 else: 

129 issues.append(ParseIssue("warning", "sub_type", "Missing sub_type (segment 1)")) 

130 

131 # Parse material or type-category (segment 2) 

132 if len(segments) > 2 and segments[2]: 

133 pp = self._parse_material_field(pp, segments[2], issues) 

134 

135 # Parse part code (segment 3) 

136 if len(segments) > 3 and segments[3]: 

137 pp = self._parse_code_field(pp, segments[3], issues) 

138 

139 # Parse dimensions (segment 4) 

140 if len(segments) > 4 and segments[4]: 

141 pp = self._parse_dimension_field(pp, segments[4], issues) 

142 

143 # Parse surface/brand/notes (segment 5+) 

144 if len(segments) > 5: 

145 pp = self._parse_tail_segments(pp, segments[5:], issues) 

146 

147 # Determine overall status 

148 if not pp.category and not pp.sub_type: 

149 status = ParseStatus.FAILED 

150 elif not pp.sub_type: 

151 status = ParseStatus.FAILED # Only category without sub_type → FAILED 

152 elif not pp.category: 

153 status = ParseStatus.PARTIAL # Has sub_type but no category 

154 elif not pp.material and pp.sub_type in _MACHINED_SUBTYPES: 

155 status = ParseStatus.PARTIAL 

156 elif not pp.dimensions_raw and pp.sub_type in _MACHINED_SUBTYPES: 

157 status = ParseStatus.PARTIAL 

158 elif any(i.severity == "error" for i in issues): 

159 status = ParseStatus.PARTIAL 

160 else: 

161 status = ParseStatus.SUCCESS 

162 

163 return DescriptionParseResult( 

164 parsed_part=pp, 

165 status=status, 

166 issues=issues, 

167 raw_segments=segments, 

168 )

170 # -- Segment parsers -- 

171 

172 def _classify_part(self, pp: ParsedPart, sub_type: str) -> ParsedPart: 

173 """Determine is_quotable based on sub_type.""" 

174 if sub_type in _MACHINED_SUBTYPES: 

175 pp.is_quotable = True 

176 elif sub_type in _ELECTRICAL_SUBTYPES or sub_type in _MECHANICAL_SUBTYPES: 

177 pp.is_quotable = False 

178 # Unknown sub_type → not quotable by default 

179 return pp 

180 

181 def _parse_material_field( 

182 self, pp: ParsedPart, raw: str, issues: list[ParseIssue] 

183 ) -> ParsedPart: 

184 """Parse segment 2 as material (加工件) or category (外購件).""" 

185 if pp.sub_type in _MACHINED_SUBTYPES: 

186 pp.material = raw 

187 # Normalize 

188 result = normalize_material(raw) 

189 if result.normalized: 

190 if result.confidence < 0.9: 

191 issues.append(ParseIssue( 

192 "warning", "material", 

193 f"Material '{raw}' normalized to '{result.normalized}' " 

194 f"(confidence={result.confidence:.0%}): {result.note}" 

195 )) 

196 else: 

197 issues.append(ParseIssue( 

198 "warning", "material", 

199 f"Cannot normalize material: '{raw}'. {result.note}" 

200 )) 

201 elif pp.sub_type in _ELECTRICAL_SUBTYPES: 

202 pp.spec = raw # e.g. "控制類" 

203 elif pp.sub_type in _MECHANICAL_SUBTYPES: 

204 pp.material = raw # e.g. "鋁型材" 

205 return pp 

206 

207 def _parse_code_field( 

208 self, pp: ParsedPart, raw: str, issues: list[ParseIssue] 

209 ) -> ParsedPart: 

210 """Parse segment 3 as part code or sub-category.""" 

211 if pp.sub_type in _MACHINED_SUBTYPES: 

212 pp.part_code = raw 

213 elif pp.sub_type in _ELECTRICAL_SUBTYPES: 

214 if pp.spec: 

215 pp.spec = f"{pp.spec};{raw}" 

216 else: 

217 pp.spec = raw 

218 elif pp.sub_type in _MECHANICAL_SUBTYPES: 

219 # Could be dimensions, code, or prefixed field 

220 if _looks_like_dimension(raw): 

221 pp.dimensions_raw = raw 

222 elif raw.startswith("圖號:") or raw.startswith("图号:"): 

223 pp.part_code = raw.split(":", 1)[-1].strip() 

224 elif raw.startswith("型號:") or raw.startswith("型号:"): 

225 pp.model_number = raw.split(":", 1)[-1].strip() 

226 elif raw.startswith("品牌:"): 

227 pp.brand = raw.split(":", 1)[-1].strip() 

228 else: 

229 pp.part_code = raw 

230 return pp 

231 

232 def _parse_dimension_field( 

233 self, pp: ParsedPart, raw: str, issues: list[ParseIssue] 

234 ) -> ParsedPart: 

235 """Parse segment 4 as dimensions, specs, or prefixed fields.""" 

236 if pp.sub_type in _MACHINED_SUBTYPES: 

237 pp.dimensions_raw = raw 

238 dim_result = parse_dimension(raw) 

239 if dim_result.status == DimensionStatus.FAILED: 

240 issues.append(ParseIssue( 

241 "warning", "dimensions", 

242 f"Cannot parse dimension: '{raw}'. {dim_result.issues}" 

243 )) 

244 elif dim_result.status == DimensionStatus.PARTIAL: 

245 issues.append(ParseIssue( 

246 "warning", "dimensions", 

247 f"Partial dimension parse: {dim_result.issues}" 

248 )) 

249 elif pp.sub_type in _ELECTRICAL_SUBTYPES: 

250 if raw.startswith("型號:") or raw.startswith("型号:"): 

251 pp.model_number = raw.split(":", 1)[-1].strip() 

252 elif raw.startswith("品牌:"): 

253 pp.brand = raw.split(":", 1)[-1].strip() 

254 else: 

255 existing = pp.spec or "" 

256 pp.spec = f"{existing};{raw}" if existing else raw 

257 elif pp.sub_type in _MECHANICAL_SUBTYPES: 

258 if raw.startswith("圖號:") or raw.startswith("图号:"): 

259 pp.part_code = raw.split(":", 1)[-1].strip() 

260 elif raw.startswith("型號:") or raw.startswith("型号:"): 

261 pp.model_number = raw.split(":", 1)[-1].strip() 

262 elif raw.startswith("品牌:"): 

263 pp.brand = raw.split(":", 1)[-1].strip() 

264 else: 

265 pp.dimensions_raw = raw 

266 return pp 

267 

268 def _parse_tail_segments( 

269 self, pp: ParsedPart, segments: list[str], issues: list[ParseIssue] 

270 ) -> ParsedPart: 

271 """Parse remaining segments (5+) for surface treatment / brand / notes.""" 

272 for seg in segments: 

273 if not seg: 

274 continue 

275 

276 if pp.sub_type in _MACHINED_SUBTYPES: 

277 # First non-dimension tail = surface treatment 

278 if not pp.surface_treatment: 

279 pp.surface_treatment = seg 

280 # Subsequent tails = extra notes (could be tolerances, etc.) 

281 elif pp.sub_type in _ELECTRICAL_SUBTYPES: 

282 if seg.startswith("型號:") or seg.startswith("型号:"): 

283 pp.model_number = seg.split(":", 1)[-1].strip() 

284 elif seg.startswith("品牌:"): 

285 pp.brand = seg.split(":", 1)[-1].strip() 

286 else: 

287 pp.spec = f"{pp.spec};{seg}" if pp.spec else seg 

288 elif pp.sub_type in _MECHANICAL_SUBTYPES: 

289 if seg.startswith("圖號:") or seg.startswith("图号:"): 

290 pp.part_code = seg.split(":", 1)[-1].strip() 

291 elif seg.startswith("型號:") or seg.startswith("型号:"): 

292 pp.model_number = seg.split(":", 1)[-1].strip() 

293 elif seg.startswith("品牌:"): 

294 pp.brand = seg.split(":", 1)[-1].strip() 

295 elif not pp.surface_treatment: 

296 pp.surface_treatment = seg 

297 

298 return pp

305def _looks_like_dimension(text: str) -> bool: 

306 """Check if text looks like a dimension (contains * or × or starts with φ).""" 

307 import re 

308 return bool(re.search(r"[*×xXφΦ]", text) or re.match(r"^\d+$", text))
