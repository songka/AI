42 def cleanup_converted_file(self, result: DwgConversionResult) -> bool: 

43 """Delete a generated DXF while protecting the original SOLIDWORKS file.""" 

44 if not result.converted_file: 

45 return False 

46 converted = Path(result.converted_file) 

47 resolved = converted.resolve() 

48 if ( 

49 resolved == Path(result.source_file).resolve() 

50 or resolved.suffix.casefold() != ".dxf" 

51 or not resolved.is_relative_to(self._cache_dir.resolve()) 

52 ): 

53 raise ValueError("拒绝删除不属于受控转换目录的文件") 

54 existed = resolved.is_file() 

55 resolved.unlink(missing_ok=True) 

56 return existed

58 def convert(self, source_path: str | Path) -> DwgConversionResult: 

59 source = Path(source_path) 

60 base = DwgConversionResult( 

61 source_file=str(source), status=ConversionStatus.FAILED, 

62 adapter="SOLIDWORKS COM", configuration_source="windows_com", 

63 ) 

64 if source.suffix.lower() not in {".slddrw", ".sldprt"}: 

65 base.error = f"不支持的 SOLIDWORKS 文件类型：{source.suffix}" 

66 return base 

67 if not source.is_file(): 

68 base.error = "找不到 SOLIDWORKS 来源文件" 

69 return base 

70 if not self.is_available(): 

71 base.status = ConversionStatus.NOT_CONFIGURED 

72 base.error = "本机未安装或未注册 SOLIDWORKS；原生 SLD 文件必须通过合法 SOLIDWORKS 自动化接口转换" 

73 return base 

74 digest = hashlib.sha256(source.read_bytes() + b"solidworks-dxf-v1").hexdigest() 

75 target = self._cache_dir / f"{digest}.dxf" 

76 if target.is_file() and target.stat().st_size > 0: 

77 base.status = ConversionStatus.SUCCESS 

78 base.converted_file = str(target) 

79 base.cache_hit = True 

80 return base 

81 self._cache_dir.mkdir(parents=True, exist_ok=True) 

82 pending = self._cache_dir / f".{digest}.dxf" 

83 try: 

84 completed = subprocess.run( 

85 [ 

86 "powershell.exe", "-NoProfile", "-ExecutionPolicy", "Bypass", 

87 "-File", str(self._script), "-SourcePath", str(source.resolve()), 

88 "-OutputPath", str(pending.resolve()), 

89 ], 

90 capture_output=True, text=True, timeout=120, check=False, 

91 creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0), 

92 ) 

93 if completed.returncode != 0 or not pending.is_file() or pending.stat().st_size == 0: 

94 base.error = (completed.stderr or completed.stdout or "SOLIDWORKS 转换失败").strip()[:500] 

95 return base 

96 pending.replace(target) 

97 base.status = ConversionStatus.SUCCESS 

98 base.converted_file = str(target) 

99 return base 

100 except (OSError, subprocess.TimeoutExpired) as exc: 

101 base.error = f"SOLIDWORKS 转换失败：{exc}" 

102 return base 

103 finally: 

104 pending.unlink(missing_ok=True)
