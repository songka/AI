13 absolute_errors = [

18 apes = [

30 else: 

31 buckets[">30%"] += 1

33 "comparable_cases": len(comparable), 

34 "wape_pct": round(sum(absolute_errors) / historical_total * 100, 2)

36 "mae_cny": round(sum(absolute_errors) / len(absolute_errors), 2)
