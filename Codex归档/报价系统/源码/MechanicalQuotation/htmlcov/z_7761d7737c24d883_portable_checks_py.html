20def _write_report(name: str, title: str, checks: list[dict[str, Any]]) -> Path: 

21 root = _root() 

22 report_dir = root / "runtime" / "reports" 

23 report_dir.mkdir(parents=True, exist_ok=True) 

24 passed = sum(check["ok"] for check in checks) 

25 report = { 

26 "timestamp": datetime.now().isoformat(), 

27 "root": str(root), 

28 "summary": {"passed": passed, "failed": len(checks) - passed}, 

29 "checks": checks, 

30 } 

31 (report_dir / f"{name}.json").write_text( 

32 json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8" 

33 ) 

34 rows = "".join( 

35 "<tr><td>{}</td><td class='{}'>{}</td><td>{}</td></tr>".format( 

36 html.escape(str(check["name"])), 

37 "pass" if check["ok"] else "fail", 

38 "通過" if check["ok"] else "失敗", 

39 html.escape(str(check.get("detail", ""))), 

40 ) 

41 for check in checks 

42 ) 

43 document = f"""<!doctype html><html lang="zh-CN"><meta charset="utf-8"> 

44<title>{html.escape(title)}</title><style> 

45body{{font-family:'Microsoft YaHei UI',sans-serif;max-width:1000px;margin:24px auto}} 

46table{{border-collapse:collapse;width:100%}}th,td{{border:1px solid #ddd;padding:8px}} 

47th{{background:#1a5276;color:white}}.pass{{color:#17833b}}.fail{{color:#bd2222}} 

48</style><h1>{html.escape(title)}</h1><p>通過 {passed}/{len(checks)}</p> 

49<table><tr><th>检查项目</th><th>状态</th><th>详情</th></tr>{rows}</table></html>""" 

50 output = report_dir / f"{name}.html" 

51 output.write_text(document, encoding="utf-8") 

52 return output

55def run_self_check() -> int: 

56 root = _root() 

57 checks: list[dict[str, Any]] = [] 

58 

59 def check(name: str, ok: bool, detail: str) -> None: 

60 checks.append({"name": name, "ok": bool(ok), "detail": detail}) 

61 

62 pointer_path = root / "data" / "current-version-pointer.json" 

63 check("程式根目錄", root.exists(), str(root)) 

64 check("正式價格版本指標", pointer_path.exists(), str(pointer_path)) 

65 if pointer_path.exists(): 

66 pointer = json.loads(pointer_path.read_text(encoding="utf-8")) 

67 snapshot = root / "data" / pointer["snapshot_path"] 

68 check("正式价格表", snapshot.exists(), pointer["current_version"]) 

69 if snapshot.exists(): 

70 payload = json.loads(snapshot.read_text(encoding="utf-8")) 

71 canonical = json.dumps(payload["company_prices"], sort_keys=True, ensure_ascii=False) 

72 actual = hashlib.sha256(canonical.encode("utf-8")).hexdigest() 

73 check("价格表完整性校验", actual == payload.get("snapshot_sha256"), actual) 

74 

75 secret = root / "runtime" / "secrets" / "deepseek_api_key.txt" 

76 check( 

77 "DeepSeek Key 独立配置", 

78 True, 

79 ( 

80 "已配置；Key 不写入软件设置或清单" 

81 if secret.is_file() and bool(secret.read_text(encoding="utf-8").strip()) 

82 else "未配置（交付默认状态）；程序使用内置规则，用户可在系统设置中单独配置" 

83 ), 

84 ) 

85 skill_protocol = root / "docs" / "external-quotation-skill-protocol-v1.0.yaml" 

86 check( 

87 "外接 Skill 协议", 

88 skill_protocol.is_file(), 

89 "支持 HTTP、本地文件夹和 SMB 公共槽文件夹", 

90 ) 

91 settings_path = root / "config" / "user_settings.json" 

92 check("设置目录", settings_path.exists(), "密钥与非敏感设置分离") 

93 check("输出目录", (root / "exports").is_dir(), str(root / "exports")) 

94 oda_files = list(root.rglob("ODAFileConverter.exe")) 

95 if oda_files: 

96 settings = json.loads(settings_path.read_text(encoding="utf-8")) 

97 configured = Path(settings.get("dwg_converter_path", "")) 

98 resolved = (settings_path.parent / configured).resolve() 

99 check( 

100 "DWG 转换器", 

101 resolved.is_file() and resolved in [item.resolve() for item in oda_files], 

102 "包内 ODA 已配置（仅限获授权电脑）", 

103 ) 

104 else: 

105 check("DWG 转换器", True, "使用电脑需另行合法安装并配置 ODA") 

106 

107 try: 

108 from quotation.infrastructure.rules.published_pricebook_loader import PublishedPricebookLoader 

109 loader = PublishedPricebookLoader(pointer_path) 

110 check("正式价格加载器", loader.loaded, loader.price_version or loader.load_error or "") 

111 except Exception as exc: 

112 check("Pricebook Loader", False, str(exc)) 

113 

114 try: 

115 import tkinter # noqa: F401 

116 check("桌面界面", True, "可载入") 

117 except Exception as exc: 

118 check("Tkinter UI", False, str(exc)) 

119 

120 try: 

121 from quotation.api.main import app 

122 check("接口服务", bool(app.openapi()["paths"]), "接口文档可生成") 

123 except Exception as exc: 

124 check("FastAPI", False, str(exc)) 

125 

126 output = _write_report("portable_self_check", "机械报价系统便携版自检报告", checks) 

127 print(f"自檢報告：{output}") 

128 return 0 if all(check["ok"] for check in checks) else 1

131def run_smoke() -> int: 

132 root = _root() 

133 checks: list[dict[str, Any]] = [] 

134 temp_dxf = root / "runtime" / "tmp" / "portable_smoke.dxf" 

135 try: 

136 import ezdxf 

137 from quotation.application.batch_excel import export_batch_excel 

138 from quotation.application.quotation_service import QuotationApplicationService 

139 

140 temp_dxf.parent.mkdir(parents=True, exist_ok=True) 

141 doc = ezdxf.new() 

142 doc.header["$INSUNITS"] = 4 

143 msp = doc.modelspace() 

144 for start, end in [((0, 0), (100, 0)), ((100, 0), (100, 50)), ((100, 50), (0, 50)), ((0, 50), (0, 0))]: 

145 msp.add_line(start, end) 

146 msp.add_text("S50C", height=5).set_placement((5, 55)) 

147 doc.saveas(temp_dxf) 

148 result = QuotationApplicationService().quote_single_file(temp_dxf) 

149 from quotation.ui.localization import display_value 

150 checks.append({ 

151 "name": "示例报价", 

152 "ok": result.quote is not None, 

153 "detail": display_value("status", result.status), 

154 }) 

155 checks.append({ 

156 "name": "13% 税务", 

157 "ok": result.tax is not None and result.tax.total_including_tax == result.tax.subtotal_excluding_tax + result.tax.tax_amount, 

158 "detail": f"未稅={result.subtotal_excluding_tax}, 含稅={result.total_including_tax}", 

159 }) 

160 excel = root / "exports" / "portable_smoke.xlsx" 

161 export_batch_excel([result], excel) 

162 checks.append({"name": "批量 Excel", "ok": excel.exists() and excel.stat().st_size > 0, "detail": str(excel)}) 

163 except Exception as exc: 

164 checks.append({"name": "Demo smoke", "ok": False, "detail": str(exc)}) 

165 finally: 

166 temp_dxf.unlink(missing_ok=True) 

167 

168 output = _write_report("portable_demo_smoke", "机械报价系统便携版功能检查报告", checks) 

169 print(f"Smoke 報告：{output}") 

170 return 0 if checks and all(check["ok"] for check in checks) else 1
